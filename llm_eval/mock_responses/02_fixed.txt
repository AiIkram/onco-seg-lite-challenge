import os, sys, random
import numpy as np, pandas as pd, torch, torch.nn as nn, torch.nn.functional as F
from PIL import Image
sys.path.insert(0, ".")
from competition.metrics import rle_encode
random.seed(25); np.random.seed(25); torch.manual_seed(25)
P = "data/public"
def load(p): return np.asarray(Image.open(p).convert("L"), dtype=np.float32) / 255.0
meta = pd.read_csv(f"{P}/train_metadata.csv", dtype=str)
X = torch.tensor(np.stack([load(f"{P}/train/images/{i}.png") for i in meta.image_id]))[:, None]
Y = torch.tensor(np.stack([load(f"{P}/train/masks/{i}.png") > 0.5 for i in meta.image_id]).astype(np.float32))[:, None]
X, Y = F.interpolate(X, 64, mode="bilinear"), F.interpolate(Y, 64)
class Net(nn.Module):
    def __init__(s):
        super().__init__()
        c = lambda i, o: nn.Sequential(nn.Conv2d(i, o, 3, padding=1), nn.BatchNorm2d(o), nn.ReLU())
        s.e1, s.e2, s.e3 = c(1, 16), c(16, 32), c(32, 64)
        s.d2, s.d1, s.h = c(96, 32), c(48, 16), nn.Conv2d(16, 1, 1)
    def forward(s, x):
        a = s.e1(x); b = s.e2(F.max_pool2d(a, 2)); c_ = s.e3(F.max_pool2d(b, 2))
        b = s.d2(torch.cat([F.interpolate(c_, scale_factor=2), b], 1))
        a = s.d1(torch.cat([F.interpolate(b, scale_factor=2), a], 1))
        return s.h(a)
m = Net(); opt = torch.optim.Adam(m.parameters(), 3e-3)
for ep in range(2):
    perm = torch.randperm(len(X))
    for j in range(0, len(X), 16):
        idx = perm[j:j + 16]; out = m(X[idx]); p = torch.sigmoid(out)
        loss = F.binary_cross_entropy_with_logits(out, Y[idx]) + 1 - (2 * (p * Y[idx]).sum() + 1) / (p.sum() + Y[idx].sum() + 1)
        opt.zero_grad(); loss.backward(); opt.step()
    print("epoch", ep, float(loss))
ids = pd.read_csv(f"{P}/test_ids.csv", dtype=str).image_id.tolist()
T = F.interpolate(torch.tensor(np.stack([load(f"{P}/test/images/{i}.png") for i in ids]))[:, None], 64, mode="bilinear")
m.eval()
with torch.no_grad():
    pr = F.interpolate(torch.sigmoid(m(T)), 256, mode="bilinear")[:, 0].numpy()
rles = []
for p in pr:
    mk = (p > 0.5).astype(np.uint8)
    rles.append(rle_encode(mk) if mk.sum() >= 100 else "")
pd.DataFrame({"image_id": ids, "rle": rles}).to_csv("predictions.csv", index=False)
print("wrote predictions.csv")
